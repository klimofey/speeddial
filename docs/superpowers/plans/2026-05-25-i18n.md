# Internationalization (i18n) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Localize the whole UI into English, Russian, Spanish, German, French, switchable in Settings (default = browser language).

**Architecture:** A flat `t(key, vars?)` over per-language dictionaries with a module-level current locale set at the start of `Board` render (`setLang(settings.language)`); components replace literal strings with `t('key')`. English dictionary equals today's wording, so English output is byte-identical and existing tests stay green.

**Tech Stack:** TypeScript, Preact, Vitest + @testing-library/preact.

  });

    // a key only present in en still resolves via the en fallback

export const messages: Record<Lang, Dict> = { en, ru, es, de, fr };

export const LANGS: { code: Lang; label: string }[] = [
  { code: 'en', label: 'English' },
  { code: 'ru', label: 'Русский' },
  { code: 'es', label: 'Español' },
  { code: 'de', label: 'Deutsch' },
  { code: 'fr', label: 'Français' },
];

let current: Lang = 'en';

export function setLang(lang: Lang): void {
  current = lang;
}

export function t(key: string, vars?: Record<string, string | number>): string {
  const raw = messages[current][key] ?? messages.en[key] ?? key;
  if (!vars) return raw;
  return raw.replace(/\{(\w+)\}/g, (_, k) => String(vars[k] ?? ''));
}

export function detectLang(locale: string = typeof navigator !== 'undefined' ? navigator.language : 'en'): Lang {
  const code = (locale || 'en').slice(0, 2).toLowerCase();
  return (['en', 'ru', 'es', 'de', 'fr'] as const).includes(code as Lang) ? (code as Lang) : 'en';
}
